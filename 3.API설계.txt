게시글 관련

	-> GET : /insert
 - 게시글 수정 화면 : "hasAuthority('POST_MODIFY') or hasRole('ADMIN')"
	-> GET : /modify/{postId}
 - 게시글 조회 상세 화면 : "hasRole('ADMIN') or hasAuthority('POST_READ')"
	-> GET : /{postId}
 - 게시글 생성 : "hasAuthority('POST_CREATE') or hasRole('ADMIN')"
	-> POST : 
 - 게시글 수정 : "(principal.getId().toString().equals(#data.get('writerId').toString()) and hasAuthority('POST_MODIFY')) or hasRole('ADMIN')"
	-> PUT : /{postId}
 - 게시글 삭제 : "(principal.getId().toString().equals(#data.get('writerId').toString()) and hasAuthority('POST_MODIFY')) or hasRole('ADMIN')"
	-> DELETE : /{postId}
